   187→   187→function relativeTimeLabel(daysAgo: number): string {
   188→   188→  if (daysAgo <= 0) return "Aujourd'hui";
   189→   189→  if (daysAgo === 1) return "Hier";
   190→   190→  if (daysAgo < 7) return `Il y a ${daysAgo} jours`;
   191→   191→  if (daysAgo < 30) {
   192→   192→    const weeks = Math.floor(daysAgo / 7);
   193→   193→    return weeks === 1 ? "Il y a 1 semaine" : `Il y a ${weeks} semaines`;
   194→   194→  }
   195→   195→  const months = Math.floor(daysAgo / 30);
   196→   196→  return months === 1 ? "Il y a 1 mois" : `Il y a ${months} mois`;
   197→   197→}

   246→   246→/**
   247→   247→ * ProgressRing — premium SVG circle that animates its stroke-dashoffset over
   248→   248→ * 1.2s using the `.progress-ring-circle` utility class (globals.css).
   249→   249→ */
   250→   250→function ProgressRing({
   251→   251→  percent,
   252→   252→  color,
   253→   253→  size = 72,
   254→   254→  stroke = 6,
   255→   255→}: {
   256→   256→  percent: number;
   257→   257→  color: string;
   258→   258→  size?: number;
   259→   259→  stroke?: number;
   260→   260→}) {
   261→   261→  const radius = (size - stroke) / 2;
   262→   262→  const circumference = 2 * Math.PI * radius;
   263→   263→  const clamped = Math.max(0, Math.min(100, percent));
   264→   264→  const offset = circumference - (clamped / 100) * circumference;
   265→   265→  return (
   266→   266→    <svg width={size} height={size} className="flex-shrink-0" aria-hidden>
   267→   267→      <circle
   268→   268→        cx={size / 2}
   269→   269→        cy={size / 2}
   270→   270→        r={radius}
   271→   271→        fill="none"
   272→   272→        stroke="rgba(255,255,255,0.08)"
   273→   273→        strokeWidth={stroke}
   274→   274→      />
   275→   275→      <motion.circle
   276→   276→        className="progress-ring-circle"
   277→   277→        cx={size / 2}
   278→   278→        cy={size / 2}
   279→   279→        r={radius}
   280→   280→        fill="none"
   281→   281→        stroke={color}
   282→   282→        strokeWidth={stroke}
   283→   283→        strokeLinecap="round"
   284→   284→        strokeDasharray={circumference}
   285→   285→        initial={{ strokeDashoffset: circumference }}
   286→   286→        animate={{ strokeDashoffset: offset }}
   287→   287→        transition={{ duration: 1.2, ease: "easeOut", delay: 0.15 }}
   288→   288→        style={{ filter: `drop-shadow(0 0 8px ${color}80)` }}
   289→   289→      />
   290→   290→      <text
   291→   291→        x="50%"
   292→   292→        y="50%"
   293→   293→        textAnchor="middle"
   294→   294→        dominantBaseline="central"
   295→   295→        className="font-[family-name:var(--font-poppins)] font-extrabold fill-white"
   296→   296→        style={{ fontSize: size * 0.22 }}
   297→   297→      >
   298→   298→        {Math.round(clamped)}%
   299→   299→      </text>
   300→   300→    </svg>
   301→   301→  );
   302→   302→}
