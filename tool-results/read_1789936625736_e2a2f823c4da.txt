   187→function relativeTimeLabel(daysAgo: number): string {
   188→  if (daysAgo <= 0) return "Aujourd'hui";
   189→  if (daysAgo === 1) return "Hier";
   190→  if (daysAgo < 7) return `Il y a ${daysAgo} jours`;
   191→  if (daysAgo < 30) {
   192→    const weeks = Math.floor(daysAgo / 7);
   193→    return weeks === 1 ? "Il y a 1 semaine" : `Il y a ${weeks} semaines`;
   194→  }
   195→  const months = Math.floor(daysAgo / 30);
   196→  return months === 1 ? "Il y a 1 mois" : `Il y a ${months} mois`;
   197→}

   228→function getTierForTarget(target: number): ExcludedTierKey {
   229→  if (target >= 365) return "legendary";
   230→  if (target >= 30) return "gold";
   231→  if (target >= 10) return "diamond";
   232→  if (target >= 5) return "silver";
   233→  return "bronze";

   246→/**
   247→ * ProgressRing — premium SVG circle that animates its stroke-dashoffset over
   248→ * 1.2s using the `.progress-ring-circle` utility class (globals.css).
   249→ */
   250→function ProgressRing({
   251→  percent,
   252→  color,
   253→  size = 72,
   254→  stroke = 6,
   255→}: {
   256→  percent: number;
   257→  color: string;
   258→  size?: number;
   259→  stroke?: number;
   260→}) {
   261→  const radius = (size - stroke) / 2;
   262→  const circumference = 2 * Math.PI * radius;
   263→  const clamped = Math.max(0, Math.min(100, percent));
   264→  const offset = circumference - (clamped / 100) * circumference;
   265→  return (
   266→    <svg width={size} height={size} className="flex-shrink-0" aria-hidden>
   267→      <circle
   268→        cx={size / 2}
   269→        cy={size / 2}
   270→        r={radius}
   271→        fill="none"
   272→        stroke="rgba(255,255,255,0.08)"
   273→        strokeWidth={stroke}
   274→      />
   275→      <motion.circle
   276→        className="progress-ring-circle"
   277→        cx={size / 2}
   278→        cy={size / 2}
   279→        r={radius}
   280→        fill="none"
   281→        stroke={color}
   282→        strokeWidth={stroke}
   283→        strokeLinecap="round"
   284→        strokeDasharray={circumference}
   285→        initial={{ strokeDashoffset: circumference }}
   286→        animate={{ strokeDashoffset: offset }}
   287→        transition={{ duration: 1.2, ease: "easeOut", delay: 0.15 }}
   288→        style={{ filter: `drop-shadow(0 0 8px ${color}80)` }}
   289→      />
   290→      <text
   291→        x="50%"
   292→        y="50%"
   293→        textAnchor="middle"
   294→        dominantBaseline="central"
   295→        className="font-[family-name:var(--font-poppins)] font-extrabold fill-white"
   296→        style={{ fontSize: size * 0.22 }}
   297→      >
   298→        {Math.round(clamped)}%
   299→      </text>
   300→    </svg>
   301→  );
   302→}

   359→/**
   360→ * FlipBadge — a 3D flip card for a single achievement. Front shows the icon
   361→ * + name; back shows the description + unlock state. Tap (or hover on
   362→ * desktop) to flip. Unlocked badges get a tier-aura glow; locked ones get a
   363→ * desaturated treatment via .achievement-locked.
   364→ */
   365→function FlipBadge({
   366→  achievement,
   367→  index,
   368→}: {
   369→  achievement: ComputedAchievement;
   370→  index: number;
   371→}) {
   372→  const [flipped, setFlipped] = useState(false);

   375→  const colorVar = {
   376→    ["--achievement-color" as string]: achievement.color,
   377→  } as CSSProperties;
   378→
   379→  return (
   380→    <motion.div
   381→      initial={{ opacity: 0, scale: 0.85 }}
   382→      animate={{ opacity: 1, scale: 1 }}
   383→      transition={{
   384→        delay: 0.05 + index * 0.03,
   385→        type: "spring",
   386→        stiffness: 280,
   387→        damping: 22,
   388→      }}
   389→    >

   392→          achievement.unlocked ? "" : "achievement-locked"
   393→        }`}
   394→        onClick={() => setFlipped((f) => !f)}
   395→        style={colorVar}
   396→      >
   397→        <div
   398→          className="badge-3d-inner w-full h-full"
   399→          style={flipped ? { transform: "rotateY(180deg)" } : undefined}

   482→export function AchievementsScreen() {
   483→  const t = useT();
   484→  const {
   485→    navigate,

   487→    adminStreakOverride,

   494→    forumPosts,

   496→  } = useStore();
   497→
   498→  const effectiveStreak = adminStreakOverride !== null ? adminStreakOverride : streakDays;
   499→  const currentRank = getCurrentRank(effectiveStreak);
   500→  const totalSaved = effectiveStreak * Math.round(weeklyBetAmount / 7);
   501→
   502→  const achievementContext: AchievementContext = useMemo(
   503→    () => ({
   504→      streakDays: effectiveStreak,
   505→      panicEventsCount: panicEvents.length,
   506→      resolvedPanicCount: panicEvents.filter((p) => p.resolved).length,
   507→      journalCount: journalEntries.length,
   508→      totalSaved,
   509→      meditationStreak,
   510→      testimonialsCount: testimonials.filter((t) => t.isMine).length,
   511→      forumPostsCount: forumPosts.length,
   512→      articlesRead,
   513→    }),
   514→    [
   515→      effectiveStreak,
   516→      panicEvents,
   517→      journalEntries.length,
   518→      totalSaved,
   519→      meditationStreak,
   520→      testimonials,
   521→      forumPosts.length,
   522→      articlesRead,
   523→    ]
   524→  );
   525→
   526→  // Compute progress for each special achievement
   527→  const computedAchievements = useMemo(
   528→    () =>
   529→      SPECIAL_ACHIEVEMENTS.map((a) => {
   530→        const current = a.getCurrent(achievementContext);
   531→        const unlocked = current >= a.target;
   532→        const progress = Math.min(100, (current / a.target) * 100);
   533→        return { ...a, current, unlocked, progress };
   534→      }),
   535→    [achievementContext]
   536→  );
   537→
   538→  const unlockedAchievements = computedAchievements.filter((a) => a.unlocked);
   539→  const unlockedRanksCount = unlockedRanks.length;
   540→
   541→  // ----- Task 13-d: tier filter + totals -----
   542→  const [activeTier, setActiveTier] = useState<TierKey>("all");
   543→  const filteredAchievements = useMemo(
   544→    () =>
   545→      activeTier === "all"
   546→        ? computedAchievements
   547→        : computedAchievements.filter(
   548→            (a) => getTierForTarget(a.target) === activeTier,
   549→          ),
   550→    [activeTier, computedAchievements],
   551→  );
   552→
   553→  const totalUnlocked = unlockedAchievements.length + unlockedRanksCount;
   554→  const totalAchievements = SPECIAL_ACHIEVEMENTS.length + PARCOURS_RANKS.length;
   555→  const unlockedPercent =
   556→    totalAchievements > 0 ? (totalUnlocked / totalAchievements) * 100 : 0;
   557→
   558→  const tierCounts = useMemo(() => {
   559→    const counts: Record<ExcludedTierKey, number> = {
   560→      bronze: 0,
   561→      silver: 0,
   562→      gold: 0,
   563→      diamond: 0,
   564→      legendary: 0,
   565→    };
   566→    for (const a of computedAchievements) {
   567→      counts[getTierForTarget(a.target)] += 1;
   568→    }
   569→    return counts;
   570→  }, [computedAchievements]);
   571→
   572→  // Build a "recent unlocks" list
   573→  const recentUnlocks: RecentUnlock[] = useMemo(() => {
   574→    const items: RecentUnlock[] = [];
   575→
   576→    // Ranks — derive "days ago" from streak (when streakDays first crossed requiredDays)
   577→    PARCOURS_RANKS.forEach((rank) => {
   578→      const isUnlocked =
   579→        effectiveStreak >= rank.requiredDays || unlockedRanks.includes(rank.key);
   580→      if (!isUnlocked) return;
   581→      const daysAgo = Math.max(0, effectiveStreak - rank.requiredDays);
   582→      items.push({
   583→        key: `rank-${rank.key}`,
   584→        name: t(rank.nameKey),
   585→        icon: rank.icon,
   586→        color: rank.color,
   587→        type: "rank",
   588→        daysAgo,
   589→      });
   590→    });
   591→
   592→    // Special achievements — derive a rough "days ago" from each metric
   593→    computedAchievements.forEach((a) => {
   594→      if (!a.unlocked) return;
   595→      let daysAgo = 0;
   596→      if (a.key === "premier-pas" || a.key === "perseverant" || a.key === "legende") {
   597→        daysAgo = Math.max(0, effectiveStreak - a.target);
   598→      } else if (a.key === "respirateur" || a.key === "survivant") {
   599→        // Use latest panic event as a proxy
   600→        const latest = panicEvents[0];
   601→        if (latest) {
   602→          const diff = Math.floor(
   603→            (Date.now() - new Date(latest.createdAt).getTime()) / 86400000
   604→          );
   605→          daysAgo = Math.max(0, diff);
   606→        }
